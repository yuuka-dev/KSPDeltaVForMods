```python
"""Celestial system tree construction and configuration scanning.

天体系ツリーの構築とcfg自動検索。
"""

from __future__ import annotations

import logging
import math
import shutil
from dataclasses import dataclass
from pathlib import Path

from kopdeltav.calculator import calculate_hohmann, low_orbit_altitude
from kopdeltav.models import CelestialBody
from kopdeltav.parser import parse_bodies

logger = logging.getLogger(__name__)


@dataclass
class CelestialSystem:
    """A complete celestial system with tree structure.

    天体系全体のツリー構造。

    Attributes:
        root: The root body (star) of the system.
        bodies: Lookup dict mapping body name to CelestialBody.
        home_world: The body with is_home_world=True.
    """

    root: CelestialBody
    bodies: dict[str, CelestialBody]
    home_world: CelestialBody


def build_tree(bodies: list[CelestialBody]) -> CelestialSystem:
    """Build a parent/children tree from a flat list of CelestialBody.

    フラットな天体リストから親子ツリーを構築する。

    Tree construction rules:
        1. Index all bodies by name.
        2. Link parent/children using reference_body_name.
        3. Root = body with no orbit (primary). Fallback: body whose
           reference_body_name is not in the index.
        4. Home world = body with is_home_world=True.
        5. Compute SOI for bodies where soi == 0.

    Args:
        bodies: List of CelestialBody instances to form into a tree.

    Returns:
        A CelestialSystem with linked tree.

    Raises:
        ValueError: If no home world is found (no body with is_home_world=True).
        ValueError: If no root (star) can be identified.
    """
    index: dict[str, CelestialBody] = {b.name: b for b in bodies}

    # Reset parent/children to avoid stale links from prior calls.
    for body in bodies:
        body.parent = None
        body.children = []

    # Link parent/children.
    for body in bodies:
        if body.reference_body_name and body.reference_body_name in index:
            parent = index[body.reference_body_name]
            body.parent = parent
            parent.children.append(body)

    # Identify root: prefer body with no orbit.
    root: CelestialBody | None = None
    for body in bodies:
        if body.orbit is None:
            root = body
            break

    # Fallback: body whose reference_body_name is not in the index.
    if root is None:
        for body in bodies:
            if body.reference_body_name and body.reference_body_name not in index:
                root = body
                break

    # Last resort: body with no parent link.
    if root is None:
        for body in bodies:
            if body.parent is None:
                root = body
                break

    if root is None:
        raise ValueError("Cannot identify root body (star) in the system")

    # Identify home world.
    home_world: CelestialBody | None = None
    for body in bodies:
        if body.is_home_world:
            home_world = body
            break

    if home_world is None:
        raise ValueError("No home world found (no body with is_home_world=True)")

    # Compute SOI for bodies that need it.
    for body in bodies:
        if body.soi == 0.0 and body.parent is not None and body.orbit is not None:
            body.soi = _compute_soi(body)

    return CelestialSystem(root=root, bodies=index, home_world=home_world)


def _compute_soi(body: CelestialBody) -> float:
    """Compute the sphere of influence radius for a body.

    天体の影響圏半径を計算する。SOI = a × (m/M)^(2/5)

    Args:
        body: The body to compute SOI for. Must have parent and orbit.

    Returns:
        SOI radius [m].
    """
    if body.parent is None or body.orbit is None:
        return 0.0
    a = body.orbit.semi_major_axis
    mass_ratio = body.mu / body.parent.mu
    return a * math.pow(mass_ratio, 2.0 / 5.0)


def sort_by_transfer_dv(
    origin: CelestialBody,
    targets: list[CelestialBody],
) -> list[tuple[CelestialBody, float]]:
    """Sort target bodies by Hohmann transfer ΔV from origin (ascending).

    ホーマン遷移ΔV昇順でターゲット天体をソートする。

    Both origin and targets must orbit the same parent body.
    Uses calculate_hohmann with the parent body's mu.

    Args:
        origin: The departure body.
        targets: List of target bodies to sort.

    Returns:
        List of (body, transfer_dv) tuples sorted by transfer_dv ascending.
    """
    if origin.parent is None or origin.orbit is None:
        return [(t, 0.0) for t in targets]

    parent = origin.parent
    origin_alt = low_orbit_altitude(origin)
    results: list[tuple[CelestialBody, float]] = []

    for target in targets:
        if target.orbit is None:
            results.append((target, float("inf")))
            continue
        try:
            # Hohmann transfer in the parent's gravity field.
            origin_parking_alt = origin.orbit.semi_major_axis - parent.radius
            target_sma = target.orbit.semi_major_axis
            if target_sma <= origin.orbit.semi_major_axis:
                # Inner transfer: swap and negate is not needed, just use abs
                r1 = min(origin.orbit.semi_major_axis, target_sma)
                r2 = max(origin.orbit.semi_major_axis, target_sma)
                hohmann = calculate_hohmann(parent, r1 - parent.radius, r2)
            else:
                hohmann = calculate_hohmann(
                    parent,
                    origin.orbit.semi_major_axis - parent.radius,
                    target_sma,
                )
            results.append((target, hohmann.total_dv))
        except (ValueError, ZeroDivisionError):
            results.append((target, float("inf")))

    results.sort(key=lambda x: x[1])
    return results


def scan_configs(
    gamedata_path: Path,
    output_dir: Path | None = None,
    exclude_dirs: frozenset[str] = frozenset({"kopernicus"}),
) -> CelestialSystem:
    """Recursively scan a GameData directory for .cfg files and build a system.

    GameDataディレクトリを再帰検索し、天体系を構築する。

    Steps:
        1. Walk gamedata_path recursively.
        2. Skip directories whose lowercase name is in exclude_dirs.
        3. Parse each .cfg file with parse_bodies().
        4. Copy .cfg files that yielded bodies to output_dir.
        5. Build tree from all collected bodies.

    Args:
        gamedata_path: Path to KSP's GameData directory.
        output_dir: Directory to copy valid .cfg files to. None to skip copying.
        exclude_dirs: Set of directory names to exclude (case-insensitive).

    Returns:
        A CelestialSystem built from all discovered bodies.

    Raises:
        FileNotFoundError: If gamedata_path does not exist.
        ValueError: If no bodies are found, or tree construction fails.
    """
    if not gamedata_path.is_dir():
        raise FileNotFoundError(f"GameData path does not exist: {gamedata_path}")

    all_bodies: list[CelestialBody] = []
    exclude_lower = {d.lower() for d in exclude_dirs}

    for cfg_path in _walk_cfg_files(gamedata_path, exclude_lower):
        try:
            source = cfg_path.read_text(encoding="utf-8", errors="replace")
        except OSError as exc:
            logger.warning("Failed to read %s: %s", cfg_path, exc)
            continue

        bodies = parse_bodies(source)
        if not bodies:
            continue

        logger.info("Found %d body(ies) in %s", len(bodies), cfg_path)
        all_bodies.extend(bodies)

        if output_dir is not None:
            _copy_cfg(cfg_path, output_dir)

    if not all_bodies:
        raise ValueError(f"No celestial bodies found in {gamedata_path}")

    return build_tree(all_bodies)


def _walk_cfg_files(
    root: Path,
    exclude_lower: set[str],
) -> list[Path]:
    """Walk directory tree collecting .cfg files, skipping excluded dirs.

    ディレクトリツリーを走査し.cfgファイルを収集する。

    Args:
        root: Root directory to walk.
        exclude_lower: Lowercased directory names to skip.

    Returns:
        List of Path objects for .cfg files found.
    """
    cfg_files: list[Path] = []
    for item in sorted(root.iterdir()):
        if item.is_dir():
            if item.name.lower() in exclude_lower:
                logger.debug("Skipping excluded directory: %s", item)
                continue
            cfg_files.extend(_walk_cfg_files(item, exclude_lower))
        elif item.suffix.lower() == ".cfg":
            cfg_files.append(item)
    return cfg_files


def _copy_cfg(src: Path, dest_dir: Path) -> None:
    """Copy a .cfg file to the output directory, preserving relative structure.

    .cfgファイルを出力ディレクトリにコピーする。

    Args:
        src: Source .cfg file path.
        dest_dir: Destination directory.
    """
    dest = dest_dir / src.name
    try:
        dest_dir.mkdir(parents=True, exist_ok=True)
        shutil.copy2(src, dest)
    except OSError as exc:
        logger.warning("Failed to copy %s to %s: %s", src, dest, exc)
```

```python
def geostationary_altitude(body: CelestialBody) -> float:
    """Calculate geostationary orbit altitude above the surface.

    静止軌道高度を計算する。r_geo = ∛(μT²/4π²)

    Args:
        body: Target celestial body. Must have rotational_period > 0.

    Returns:
        Geostationary orbit altitude above surface [m].

    Raises:
        ValueError: If rotational_period is zero or negative.
    """
    if body.rotational_period <= 0:
        raise ValueError(f"Rotation period must be positive: {body.rotational_period}")
    r_geo = (body.mu * body.rotational_period**2 / (4.0 * math.pi**2)) ** (1.0 / 3.0)
    return r_geo - body.radius


def geostationary_dv(body: CelestialBody) -> HohmannResult:
    """Calculate Hohmann transfer from low orbit to geostationary orbit.

    低軌道から静止軌道へのホーマン遷移ΔVを計算する。

    Args:
        body: Target celestial body.

    Returns:
        HohmannResult for low orbit → geostationary transfer.
    """
    lo_alt = low_orbit_altitude(body)
    geo_alt = geostationary_altitude(body)
    geo_r = body.radius + geo_alt
    return calculate_hohmann(body, lo_alt, geo_r)


def escape_dv_from_low_orbit(body: CelestialBody) -> float:
    """Calculate ΔV to escape from low orbit.

    低軌道からの脱出ΔVを計算する。

    Args:
        body: Target celestial body.

    Returns:
        Escape ΔV from low orbit [m/s].
    """
    lo_alt = low_orbit_altitude(body)
    v_circ = circular_velocity(body, lo_alt)
    v_esc = escape_velocity(body, lo_alt)
    return v_esc - v_circ


def landing_dv(body: CelestialBody) -> tuple[float, float | None]:
    """Calculate landing ΔV from low orbit.

    低軌道からの着陸ΔVを計算する。

    Returns a tuple of (powered_landing_dv, aerobrake_landing_dv).
    powered_landing_dv is approximately the orbital velocity (full retro burn).
    aerobrake_landing_dv is approximately 0 if atmosphere exists, None otherwise.

    Args:
        body: Target celestial body.

    Returns:
        Tuple of (powered_dv, aerobrake_dv). aerobrake_dv is None for
        airless bodies.
    """
    lo_alt = low_orbit_altitude(body)
    powered = circular_velocity(body, lo_alt)
    if body.atmosphere is not None:
        aerobrake: float | None = 0.0
    else:
        aerobrake = None
    return (powered, aerobrake)
```

```python
@dataclass
class DvStep:
    """A single step in a ΔV route calculation.

    ΔVルート計算の1ステップ。

    Attributes:
        label: Description of this step.
        dv: ΔV for this step [m/s].
        cumulative: Running total ΔV from surface [m/s].
        note: Optional annotation (e.g. "aerobrake possible").
    """

    label: str
    dv: float
    cumulative: float
    note: str = ""


def compute_route(
    system: CelestialSystem,
    destination: CelestialBody | None = None,
    moon: CelestialBody | None = None,
) -> list[DvStep]:
    """Compute a full ΔV route from home world surface to a destination.

    母星地表から目的地までの累積ΔVルートを計算する。

    Route scenarios:
        - destination=None → third cosmic velocity (escape star system)
        - destination=planet, moon=None → planet orbit insertion + landing
        - destination=planet, moon=satellite → satellite orbit + landing

    Args:
        system: The celestial system.
        destination: Target planet, or None for system escape.
        moon: Target moon of the destination planet, or None.

    Returns:
        List of DvStep with cumulative ΔV.
    """
    home = system.home_world
    steps: list[DvStep] = []
    cumulative = 0.0

    # Step 1: Launch to low orbit
    lo_alt = low_orbit_altitude(home)
    launch = calculate_launch(home, lo_alt)
    cumulative += launch.total_rocket
    steps.append(DvStep(
        label="低軌道投入",
        dv=launch.total_rocket,
        cumulative=cumulative,
        note=f"高度 {lo_alt:.0f} m",
    ))

    # Step 2: Escape home world
    esc_dv = escape_dv_from_low_orbit(home)
    cumulative += esc_dv
    steps.append(DvStep(
        label="母星脱出",
        dv=esc_dv,
        cumulative=cumulative,
    ))

    if destination is None:
        # Third cosmic velocity: escape from star system
        if home.parent is not None and home.orbit is not None:
            parent_esc = escape_dv_from_low_orbit(home.parent) if home.parent.orbit is not None else 0.0
            # Simplified: escape from star = hohmann to infinity from home's orbit
            star_esc = escape_velocity(home.parent, home.orbit.semi_major_axis - home.parent.radius) - circular_velocity(home.parent, home.orbit.semi_major_axis - home.parent.radius)
            cumulative += star_esc
            steps.append(DvStep(
                label="恒星系脱出",
                dv=star_esc,
                cumulative=cumulative,
            ))
        return steps

    # Step 3: Hohmann transfer to destination planet
    if home.parent is not None and home.orbit is not None and destination.orbit is not None:
        origin_sma = home.orbit.semi_major_axis
        target_sma = destination.orbit.semi_major_axis
        parent = home.parent
        r1 = min(origin_sma, target_sma)
        r2 = max(origin_sma, target_sma)
        hohmann = calculate_hohmann(parent, r1 - parent.radius, r2)
        cumulative += hohmann.total_dv
        steps.append(DvStep(
            label=f"{destination.display_name}遷移",
            dv=hohmann.total_dv,
            cumulative=cumulative,
        ))

    # Step 4: Orbit insertion at destination
    dest_lo_alt = low_orbit_altitude(destination)
    dest_v_circ = circular_velocity(destination, dest_lo_alt)
    # Capture burn ≈ escape_dv (reverse of escape)
    dest_capture = escape_dv_from_low_orbit(destination)
    cumulative += dest_capture
    steps.append(DvStep(
        label=f"{destination.display_name}軌道投入",
        dv=dest_capture,
        cumulative=cumulative,
    ))

    if moon is not None:
        # Step 5: Transfer to moon
        if moon.orbit is not None:
            moon_hohmann = calculate_hohmann(destination, dest_lo_alt, moon.orbit.semi_major_axis)
            cumulative += moon_hohmann.total_dv
            steps.append(DvStep(
                label=f"{moon.display_name}遷移",
                dv=moon_hohmann.total_dv,
                cumulative=cumulative,
            ))

        # Step 6: Moon landing
        moon_powered, moon_aero = landing_dv(moon)
        cumulative += moon_powered
        note = ""
        if moon_aero is not None:
            note = f"エアロブレーキ時: {moon_aero:.0f} m/s"
        steps.append(DvStep(
            label=f"{moon.display_name}着陸",
            dv=moon_powered,
            cumulative=cumulative,
            note=note,
        ))
    else:
        # Step 5: Land on destination planet
        dest_powered, dest_aero = landing_dv(destination)
        cumulative += dest_powered
        note = ""
        if dest_aero is not None:
            note = f"エアロブレーキ時: {dest_aero:.0f} m/s"
        steps.append(DvStep(
            label=f"{destination.display_name}着陸",
            dv=dest_powered,
            cumulative=cumulative,
            note=note,
        ))

    return steps
```

```python
def _interactive_mode(system: CelestialSystem) -> None:
    """Run the interactive ΔV map navigator."""
    _print_home_info(system)

    home = system.home_world
    root = system.root

    # Get planets (root's children excluding home world)
    planets = [c for c in root.children if c.name != home.name]
    sorted_planets = sort_by_transfer_dv(home, planets)

    print(f"\n遷移先を選択:")
    for i, (planet, dv) in enumerate(sorted_planets, 1):
        print(f"  {i}) {planet.display_name:16s} (ΔV: {_fmt_number(dv, 1)} m/s)")
    print(f"  [Enter] 第三宇宙速度（恒星系脱出）")

    choice = input("\n> ").strip()

    if not choice:
        # Third cosmic velocity
        route = compute_route(system)
        print(f"\n=== 恒星系脱出 ===")
        _print_route(route)
        return

    try:
        idx = int(choice) - 1
        if 0 <= idx < len(sorted_planets):
            selected_planet = sorted_planets[idx][0]
        else:
            print("無効な選択です。")
            return
    except ValueError:
        print("無効な入力です。")
        return

    # Show planet's moons
    moons = selected_planet.children
    if moons:
        sorted_moons = sort_by_transfer_dv(selected_planet, moons)
        print(f"\n=== {home.display_name} → {selected_planet.display_name} ===")
        route_to_planet = compute_route(system, destination=selected_planet)
        _print_route(route_to_planet[:-1])  # exclude landing step for now

        print(f"\n衛星を選択:")
        for i, (moon, dv) in enumerate(sorted_moons, 1):
            print(f"  {i}) {moon.display_name:16s} (ΔV: {_fmt_number(dv, 1)} m/s)")
        print(f"  [Enter] {selected_planet.display_name}に着陸")

        moon_choice = input("\n> ").strip()
        if not moon_choice:
            route = compute_route(system, destination=selected_planet)
        else:
            try:
                moon_idx = int(moon_choice) - 1
                if 0 <= moon_idx < len(sorted_moons):
                    selected_moon = sorted_moons[moon_idx][0]
                    route = compute_route(system, destination=selected_planet, moon=selected_moon)
                else:
                    print("無効な選択です。")
                    return
            except ValueError:
                print("無効な入力です。")
                return
    else:
        route = compute_route(system, destination=selected_planet)

    print(f"\n=== ΔVルート ===")
    _print_route(route)

def _load_interactive() -> None:
    """Load from celestial_body/ and start interactive mode."""
    if not CELESTIAL_BODY_DIR.is_dir():
        print("celestial_body/ が見つかりません。")
        print("先に --scan で GameData をスキャンしてください。")
        path_input = input("GameData パス (空白で終了): ").strip()
        if not path_input:
            sys.exit(0)
        _scan_mode(Path(path_input))
        return

    all_bodies: list[CelestialBody] = []
    for cfg_file in sorted(CELESTIAL_BODY_DIR.glob("*.cfg")):
        try:
            source = cfg_file.read_text(encoding="utf-8")
            all_bodies.extend(parse_bodies(source))
        except Exception as exc:
            print(f"警告: {cfg_file.name} の読み込みに失敗: {exc}", file=sys.stderr)

    if not all_bodies:
        print("天体が見つかりません。--scan で再スキャンしてください。")
        sys.exit(1)

    system = build_tree(all_bodies)
    _interactive_mode(system)


def main() -> None:
    """Entry point for the CLI."""
    if len(sys.argv) < 2:
        print("使用方法:")

    if sys.argv[1] == "--scan":
        if len(sys.argv) < 3:
            print("エラー: GameDataパスを指定してください。")
            sys.exit(1)
        _scan_mode(Path(sys.argv[2]))
    elif sys.argv[1] == "--interactive":
        _load_interactive()
    else:
        # Existing single-cfg mode
        cfg_path = Path(sys.argv[1])
        if not cfg_path.exists():
            print(f"エラー: ファイルが見つかりません: {cfg_path}", file=sys.stderr)

            source = cfg_path.read_text(encoding="utf-8")

            sys.exit(1)
        try:
            bodies = parse_bodies(source)
        except Exception as exc:
            print(f"エラー: 設定ファイルの解析に失敗しました: {exc}", file=sys.stderr)
            sys.exit(1)
        if not bodies:
            print("警告: 天体が見つかりませんでした。", file=sys.stderr)
            sys.exit(1)
        print(f"KSPDeltaVForMods — {cfg_path.name} ({len(bodies)} 天体)")
        print("=" * 50)
        for body in bodies:
            try:
                _print_body(body)
            except Exception as exc:
                print(f"  エラー: {body.name} の計算に失敗しました: {exc}", file=sys.stderr)


if __name__ == "__main__":
```
